// Point to localhost for native binary execution with Dockerized Postgres
const dbConnStr = "postgres://postgres:password@localhost:5432/url_shortener?sslmode=disable"

func main() {
    db, err := sql.Open("postgres", dbConnStr)
    if err != nil {
        panic(err)
    }
    defer db.Close()

    e := echo.New()
    e.POST("/url", putURLHandler(db))
    e.GET("/:id", getURLHandler(db))

  <CodeBlock

    code={`# 1. Stop PostgreSQL container completely
docker compose stop postgres

# 2. Confirm database is offline

  />

  <Callout type="aha" title="Why This Works">
    When your Go binary queries <code>localhost:5432</code>, Keploy&apos;s eBPF proxy intercepts the TCP connection at the kernel level and serves the exact pre-recorded PostgreSQL handshake and SQL response from <code>keploy/test-set-0/mocks.yaml</code>. All 3 tests pass in seconds with zero live database dependencies!
  </Callout>
